#### 1.4.3 主题切换 Hook

```typescript
// src/hooks/useTheme.ts
import { useState, useEffect, useCallback } from "react";

export type Theme = "light" | "dark" | "dracula" | "nord" | "monokai" | "glass";

const THEME_KEY = "plan-todos-theme";

export function useTheme() {
  const [theme, setThemeState] = useState<Theme>("light");

  // 初始化时从 localStorage 读取
  useEffect(() => {
    const stored = localStorage.getItem(THEME_KEY) as Theme;
    if (stored && isValidTheme(stored)) {
      setTheme(stored);
    } else {
      // 检测系统偏好
      const prefersDark = window.matchMedia(
        "(prefers-color-scheme: dark)",
      ).matches;
      setTheme(prefersDark ? "dark" : "light");
    }
  }, []);

  const setTheme = useCallback((newTheme: Theme) => {
    setThemeState(newTheme);
    document.documentElement.setAttribute("data-theme", newTheme);
    localStorage.setItem(THEME_KEY, newTheme);
  }, []);

  return { theme, setTheme };
}

function isValidTheme(t: string): t is Theme {
  return ["light", "dark", "dracula", "nord", "monokai", "glass"].includes(t);
}
```

#### 1.4.4 主题选择器组件

```tsx
// src/components/ui/ThemeSelector.tsx
import { useTheme, Theme } from "@/hooks/useTheme";

const THEMES: { id: Theme; name: string; preview: string }[] = [
  { id: "light", name: "Light", preview: "#F0FDFA" },
  { id: "dark", name: "Dark", preview: "#0F172A" },
  { id: "dracula", name: "Dracula", preview: "#282A36" },
  { id: "nord", name: "Nord", preview: "#2E3440" },
  { id: "monokai", name: "Monokai", preview: "#272822" },
  { id: "glass", name: "Glass", preview: "linear-gradient" },
];

export function ThemeSelector() {
  const { theme, setTheme } = useTheme();

  return (
    <div className="grid grid-cols-3 gap-3">
      {THEMES.map((t) => (
        <button
          key={t.id}
          onClick={() => setTheme(t.id)}
          className={`
            flex flex-col items-center p-3 rounded-lg border-2 transition-all
            ${
              theme === t.id
                ? "border-[var(--color-primary)] bg-[var(--color-primary)]/10"
                : "border-[var(--color-border)] hover:border-[var(--color-primary)]/50"
            }
          `}
        >
          <div
            className="w-12 h-12 rounded-md mb-2"
            style={{
              background: t.preview,
              border: "1px solid var(--color-border)",
            }}
          />
          <span className="text-sm text-[var(--color-text)]">{t.name}</span>
        </button>
      ))}
    </div>
  );
}
```

#### 主题切换动画设计

```
触发条件: 用户点击主题卡片选择新主题

动画效果:
- 全局颜色渐变过渡 (300ms ease-in-out)
- 背景色从旧主题平滑过渡到新主题
- 文字颜色在过渡中途切换 (150ms)
- 边框颜色同步过渡

实现方式:
- 在 globals.css 中为 [data-theme] 添加 transition
- 推荐: CSS transition on CSS variables (性能更好)

示例代码:
[data-theme] {
  transition:
    background-color 0.3s ease,
    color 0.15s ease 0.15s,
    border-color 0.3s ease;
}

注意事项:
- 不要使用 framer-motion，CSS transition 性能更好
- 确保过渡时间适中，太长会影响体验
- 颜色变化应该自然
```
