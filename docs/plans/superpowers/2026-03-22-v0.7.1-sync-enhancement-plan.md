```rust
// src-tauri/src/sync/state.rs
use serde::Serialize;
use std::sync::atomic::{AtomicU8, AtomicU32, Ordering};
use tokio::sync::RwLock;

#[derive(Debug, Clone, Serialize)]
#[serde(tag = "type", rename_all = "lowercase")]
pub enum SyncStatus {
    Idle,
    Syncing { progress: u32 },
    Error { message: String },
}

pub struct SyncState {
    status: AtomicU8,
    progress: AtomicU32,
    error_message: RwLock<Option<String>>,
}

impl Default for SyncState {
    fn default() -> Self {
        Self::new()
    }
}

impl SyncState {
    pub fn new() -> Self {
        Self {
            status: AtomicU8::new(0),
            progress: AtomicU32::new(0),
            error_message: RwLock::new(None),
        }
    }

    pub fn set_syncing(&self, progress: u32) {
        self.status.store(1, Ordering::SeqCst);
        self.progress.store(progress.min(100), Ordering::SeqCst);
    }

    pub fn set_idle(&self) {
        self.status.store(0, Ordering::SeqCst);
        self.progress.store(100, Ordering::SeqCst);
        if let Ok(mut guard) = self.error_message.try_write() {
            *guard = None;
        }
    }

    pub fn set_error(&self, message: &str) {
        self.status.store(2, Ordering::SeqCst);
        if let Ok(mut guard) = self.error_message.try_write() {
            *guard = Some(message.to_string());
        }
    }

    pub fn get_status(&self) -> SyncStatus {
        match self.status.load(Ordering::SeqCst) {
            0 => SyncStatus::Idle,
            1 => SyncStatus::Syncing {
                progress: self.progress.load(Ordering::SeqCst),
            },
            2 => {
                let msg = self
                    .error_message
                    .try_read()
                    .ok()
                    .and_then(|g| g.clone())
                    .unwrap_or_default();
                SyncStatus::Error { message: msg }
            }
            _ => SyncStatus::Idle,
        }
    }
}
```

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]

    Upload,

        }
        SyncDirection::Download => {
            state.set_syncing(50);
            // sync_download().await?

        }
        SyncDirection::Both => {

#[tauri::command]
pub async fn resolve_conflicts_batch(resolutions: Vec<Resolution>) -> Result<usize, String> {
    let mut count = 0;
    for resolution in resolutions {
        match resolve_conflict(resolution).await {
            Ok(_) => count += 1,
            Err(e) => log::warn!("Failed to resolve conflict: {}", e),
        }
    }
    Ok(count)
}

#[tauri::command]
pub async fn resolve_all_with_strategy(strategy: ResolutionStrategy) -> Result<usize, String> {
    let conflicts = get_conflicts().await?;
    let resolutions: Vec<Resolution> = conflicts
        .iter()
        .map(|c| Resolution {
            conflict_id: c.id.clone(),
            strategy: strategy.clone(),
            merged_data: None,
        })
        .collect();

    resolve_conflicts_batch(resolutions).await
}

// src/components/features/ConflictList.tsx
"use client";

import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";
import { getConflicts, resolveConflict, resolveAllWithStrategy, type ResolutionStrategy } from "@/lib/api/conflict";
import { ConflictCard } from "./ConflictCard";
import { Button } from "@/components/ui/Button";
import { Skeleton } from "@/components/ui/Skeleton";

interface ConflictListProps {
  onResolved?: () => void;
}

export function ConflictList({ onResolved }: ConflictListProps) {
  const queryClient = useQueryClient();

  const { data: conflicts, isLoading } = useQuery({
    queryKey: ["conflicts"],
    queryFn: getConflicts,
  });

  const resolveMutation = useMutation({
    mutationFn: resolveConflict,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["conflicts"] });
      toast.success("冲突已解决");
      onResolved?.();
    },
    onError: (error) => {
      toast.error(`解决失败: ${error}`);
    },
  });

  const resolveAllMutation = useMutation({
    mutationFn: resolveAllWithStrategy,
    onSuccess: (count) => {
      queryClient.invalidateQueries({ queryKey: ["conflicts"] });
      toast.success(`已解决 ${count} 个冲突`);
      onResolved?.();
    },
  });

  if (isLoading) return <Skeleton className="h-32" />;
  if (!conflicts?.length) {
    return (
      <div className="text-center py-8 text-[var(--color-text-secondary)]">
        没有同步冲突
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h3 className="text-lg font-medium">同步冲突 ({conflicts.length})</h3>
        <div className="flex gap-2">
          <Button
            variant="outline"
            size="sm"
            onClick={() => resolveAllMutation.mutate("keep_local")}
            disabled={resolveAllMutation.isPending}
          >
            全部保留本地
          </Button>
          <Button
            variant="outline"
            size="sm"
            onClick={() => resolveAllMutation.mutate("keep_remote")}
            disabled={resolveAllMutation.isPending}
          >
            全部保留远程
          </Button>
        </div>
      </div>

      <div className="space-y-3">
        {conflicts.map((conflict) => (
          <ConflictCard
            key={conflict.id}
            conflict={conflict}
            onResolve={(strategy) =>
              resolveMutation.mutate({
                conflict_id: conflict.id,
                strategy,
              })
            }
            isResolving={resolveMutation.isPending}
          />
        ))}
      </div>
    </div>
  );
}
```

          <Badge variant="outline">{conflict.entity_type}</Badge>
          <span className="font-medium">{title}</span>
        </div>
        <Badge variant="destructive">{conflictTypeLabels[conflict.conflict_type]}</Badge>
      </div>

      <div className="text-sm text-[var(--color-text-secondary)] mb-3">
        本地: {format(new Date(conflict.local_modified_at), "yyyy-MM-dd HH:mm")} |
        远程: {format(new Date(conflict.remote_modified_at), "yyyy-MM-dd HH:mm")}
      </div>

      {showDiff && (
        <div className="grid grid-cols-2 gap-4 mb-3 p-3 bg-[var(--color-muted)] rounded text-xs">
          <div>
            <h4 className="font-medium mb-1">本地版本</h4>
            <pre className="overflow-auto max-h-32">
              {JSON.stringify(conflict.local_version, null, 2)}
            </pre>
          </div>
          <div>
            <h4 className="font-medium mb-1">远程版本</h4>
            <pre className="overflow-auto max-h-32">
              {JSON.stringify(conflict.remote_version, null, 2)}
            </pre>
          </div>
        </div>
      )}

      <div className="flex gap-2">
        <Button size="sm" onClick={() => onResolve("keep_local")} disabled={isResolving}>
          保留本地
        </Button>
        <Button size="sm" onClick={() => onResolve("keep_remote")} disabled={isResolving}>
          保留远程
        </Button>
        {conflict.conflict_type === "both_modified" && (
          <Button size="sm" variant="outline" onClick={() => onResolve("merge")} disabled={isResolving}>
            合并
          </Button>
        )}
        <Button size="sm" variant="ghost" onClick={() => setShowDiff(!showDiff)}>
          {showDiff ? "隐藏详情" : "查看详情"}
        </Button>
      </div>
    </div>
  );
}
```
