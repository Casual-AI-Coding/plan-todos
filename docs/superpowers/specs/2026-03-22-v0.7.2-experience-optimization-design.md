## 功能 1: 拖拽排序

### 1.1 需求描述

- Todo/Plan/Target 列表支持拖拽调整顺序
- 拖拽时显示占位符
- 支持跨分组拖拽

### 1.2 技术设计

项目已集成 `@dnd-kit/core` 和 `@dnd-kit/sortable`，可直接复用。

#### 使用现有组件

```typescript
// 项目中已有 SortableList 组件
// src/components/features/SortableList.tsx

// 在 TodosView 中启用拖拽排序
import { SortableList, SortableItem } from '@/components/features/SortableList';

function TodosView() {
  const [items, setItems] = useState<Todo[]>([]);

  const handleReorder = async (activeId: string, overId: string) => {
    // 更新排序
    const newItems = arrayMove(items, oldIndex, newIndex);
    setItems(newItems);

    // 持久化到后端
    await updateTodoOrder(activeId, overId);
  };

  return (
    <SortableList items={items} onReorder={handleReorder}>
      {(todo) => <TodoItem todo={todo} />}
    </SortableList>
  );
}
```

#### 后端 API

```rust
// src-tauri/src/commands/todos.rs

#[tauri::command]
pub fn update_todo_order(
    db: State<'_, Database>,
    id: String,
    new_order: i32,
) -> Result<(), String> {
    let conn = db.lock().map_err(|e| e.to_string())?;

    conn.execute(
        "UPDATE todos SET sort_order = ? WHERE id = ?",
        [new_order, id],
    ).map_err(|e| e.to_string())?;

    Ok(())
}

#[tauri::command]
pub fn batch_update_todo_order(
    db: State<'_, Database>,
    orders: Vec<(String, i32)>,
) -> Result<(), String> {
    let conn = db.lock().map_err(|e| e.to_string())?;

    let tx = conn.transaction().map_err(|e| e.to_string())?;

    for (id, order) in orders {
        tx.execute(
            "UPDATE todos SET sort_order = ? WHERE id = ?",
            rusqlite::params![order, id],
        ).map_err(|e| e.to_string())?;
    }

    tx.commit().map_err(|e| e.to_string())?;
    Ok(())
}
```

### 2.2 技术设计

#### ContextMenu 组件

```typescript
// src/components/ui/ContextMenu.tsx

interface ContextMenuProps {
  items: ContextMenuItem[];
  children: React.ReactNode;
}

interface ContextMenuItem {
  label: string;
  icon?: React.ReactNode;
  shortcut?: string;
  onClick: () => void;
  danger?: boolean;
  disabled?: boolean;
}

export function ContextMenu({ items, children }: ContextMenuProps) {
  return (
    <ContextMenuRoot>
      <ContextMenuTrigger>{children}</ContextMenuTrigger>
      <ContextMenuContent>
        {items.map((item, idx) => (
          <ContextMenuItem key={idx} onClick={item.onClick} disabled={item.disabled}>
            {item.icon && <span className="mr-2">{item.icon}</span>}
            {item.label}
            {item.shortcut && (
              <span className="ml-auto text-xs text-muted-foreground">
                {item.shortcut}
              </span>
            )}
          </ContextMenuItem>
        ))}
      </ContextMenuContent>
    </ContextMenuRoot>
  );
}
```

// src/components/features/BatchActionBar.tsx

function BatchActionBar({ entityType, selectedIds }: BatchActionBarProps) {
  // 添加标签选择器
  const [showTagSelector, setShowTagSelector] = useState(false);
  const [tagMode, setTagMode] = useState<'add' | 'remove'>('add');

  const handleTagAction = async (tagIds: string[], mode: 'add' | 'remove') => {
    if (mode === 'add') {
      await bulkAddTags(entityType, selectedIds, tagIds);
    } else {
      await bulkRemoveTags(entityType, selectedIds, tagIds);
    }
    toast.success(`已${mode === 'add' ? '添加' : '移除'}标签`);
  };

  return (
    <div className="flex items-center gap-3">
      {/* 现有按钮 */}

      {/* 标签操作 */}
      <Popover open={showTagSelector} onOpenChange={setShowTagSelector}>
        <PopoverTrigger asChild>
          <Button variant="outline" size="sm">
            <Tag className="w-4 h-4 mr-1" />
            标签
          </Button>
        </PopoverTrigger>
        <PopoverContent>
          <TagSelector
            mode={tagMode}
            onModeChange={setTagMode}
            onSelect={(tagIds) => handleTagAction(tagIds, tagMode)}
          />
        </PopoverContent>
      </Popover>
    </div>
  );
}

interface TagSelectorProps {
  mode: 'add' | 'remove';
  onModeChange: (mode: 'add' | 'remove') => void;
  onSelect: (tagIds: string[]) => void;
}

function TagSelector({ mode, onModeChange, onSelect }: TagSelectorProps) {
  const { data: tags } = useTags();
  const [selectedTags, setSelectedTags] = useState<Set<string>>(new Set());

  return (
    <div className="w-64">
      <div className="flex gap-2 mb-3">
        <Button
          size="sm"
          variant={mode === 'add' ? 'default' : 'outline'}
          onClick={() => onModeChange('add')}
        >
          添加
        </Button>
        <Button
          size="sm"
          variant={mode === 'remove' ? 'default' : 'outline'}
          onClick={() => onModeChange('remove')}
        >
          移除
        </Button>
      </div>

      <div className="space-y-2 max-h-48 overflow-auto">
        {tags?.map((tag) => (
          <label key={tag.id} className="flex items-center gap-2 cursor-pointer">
            <input
              type="checkbox"
              checked={selectedTags.has(tag.id)}
              onChange={() => {
                const newSet = new Set(selectedTags);
                selectedTags.has(tag.id) ? newSet.delete(tag.id) : newSet.add(tag.id);
                setSelectedTags(newSet);
              }}
            />
            <span
              className="w-3 h-3 rounded-full"
              style={{ backgroundColor: tag.color }}
            />
            <span>{tag.name}</span>
          </label>
        ))}
      </div>

      <Button
        className="w-full mt-3"
        onClick={() => onSelect(Array.from(selectedTags))}
        disabled={selectedTags.size === 0}
      >
        确认 ({selectedTags.size})
      </Button>
    </div>
  );
}

function CompletionAnimation({ onComplete }: { onComplete: () => void }) {
  const prefersReducedMotion = useMediaQuery('(prefers-reduced-motion: reduce)');

    // 无动画，直接完成

## 功能 7: 键盘导航

### 7.1 需求描述

- 上下键在列表项之间导航
- Enter 打开选中项详情
- Escape 取消选择
- Space 切换完成状态

### 7.2 技术设计

```typescript
// src/hooks/useListNavigation.ts

interface UseListNavigationOptions {
  items: { id: string }[];
  onSelect: (id: string) => void;
  onOpen: (id: string) => void;
  onToggle?: (id: string) => void;
}

export function useListNavigation({
  items,
  onSelect,
  onOpen,
  onToggle,
}: UseListNavigationOptions) {
  const [focusedIndex, setFocusedIndex] = useState(-1);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // 忽略输入框内的按键
      if (
        e.target instanceof HTMLInputElement ||
        e.target instanceof HTMLTextAreaElement
      ) {
        return;
      }

      switch (e.key) {
        case "ArrowDown":
          e.preventDefault();
          setFocusedIndex((prev) => Math.min(prev + 1, items.length - 1));
          break;
        case "ArrowUp":
          e.preventDefault();
          setFocusedIndex((prev) => Math.max(prev - 1, 0));
          break;
        case "Enter":
          if (focusedIndex >= 0) {
            onOpen(items[focusedIndex].id);
          }
          break;
        case " ":
          if (focusedIndex >= 0 && onToggle) {
            e.preventDefault();
            onToggle(items[focusedIndex].id);
          }
          break;
        case "Escape":
          setFocusedIndex(-1);
          break;
      }
    };

    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [items, focusedIndex, onOpen, onToggle]);

  // 更新选中项
  useEffect(() => {
    if (focusedIndex >= 0) {
      onSelect(items[focusedIndex].id);
    }
  }, [focusedIndex, items, onSelect]);

  return { focusedIndex, setFocusedIndex };
}
```
