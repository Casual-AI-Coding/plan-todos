## 功能 1: 同步状态追踪

### 1.1 需求描述

- 前端可查询当前同步状态
- 状态包括：空闲、同步中、错误
- 支持同步进度百分比

### 1.2 技术设计

#### 后端 - 状态管理 (`src-tauri/src/sync/engine.rs`)

```rust
use std::sync::atomic::{AtomicU8, AtomicU32, Ordering};
use tokio::sync::RwLock;

#[derive(Debug, Clone, Serialize)]
pub enum SyncStatus {
    Idle,
    Syncing { progress: u32 },
    Error { message: String },
}

pub struct SyncState {
    status: AtomicU8,  // 0: Idle, 1: Syncing, 2: Error
    progress: AtomicU32,
    error_message: RwLock<Option<String>>,
}

impl SyncState {
    pub fn new() -> Self {
        Self {
            status: AtomicU8::new(0),
            progress: AtomicU32::new(0),
            error_message: RwLock::new(None),
        }
    }

    pub fn set_syncing(&self, progress: u32) {
        self.status.store(1, Ordering::SeqCst);
        self.progress.store(progress, Ordering::SeqCst);
    }

    pub fn set_idle(&self) {
        self.status.store(0, Ordering::SeqCst);
        self.progress.store(100, Ordering::SeqCst);
    }

    pub fn set_error(&self, message: &str) {
        self.status.store(2, Ordering::SeqCst);
        if let Ok(mut guard) = self.error_message.try_write() {
            *guard = Some(message.to_string());
        }
    }

    pub fn get_status(&self) -> SyncStatus {
        match self.status.load(Ordering::SeqCst) {
            0 => SyncStatus::Idle,
            1 => SyncStatus::Syncing {
                progress: self.progress.load(Ordering::SeqCst),
            },
            2 => {
                let msg = self.error_message.try_read()
                    .ok()
                    .and_then(|g| g.clone())
                    .unwrap_or_default();
                SyncStatus::Error { message: msg }
            }
            _ => SyncStatus::Idle,
        }
    }
}
```

```rust
#[derive(Debug, Clone, Serialize)]
pub enum SyncDirection {
    Upload,
    Download,
    Both,
}

#[derive(Debug, Serialize)]
pub struct SyncResult {
    pub uploaded: usize,
    pub downloaded: usize,
    pub conflicts: usize,
    pub duration_ms: u64,
}

#[tauri::command]
pub async fn trigger_sync(
    state: State<'_, SyncState>,
    direction: SyncDirection,
) -> Result<SyncResult, String> {
    state.set_syncing(0);

    let start = std::time::Instant::now();

    // 执行同步逻辑
    let result = match direction {
        SyncDirection::Upload => sync_upload(&state).await?,
        SyncDirection::Download => sync_download(&state).await?,
        SyncDirection::Both => sync_both(&state).await?,
    };

    state.set_idle();

    Ok(SyncResult {
        uploaded: result.uploaded,
        downloaded: result.downloaded,
        conflicts: result.conflicts,
        duration_ms: start.elapsed().as_millis() as u64,
    })
}
```

#### 数据结构

```rust
// src-tauri/src/sync/conflict.rs

#[derive(Debug, Clone, Serialize)]
pub enum ConflictType {
    BothModified,      // 两边都修改了
    LocalDeleted,      // 本地删除，远程修改
    RemoteDeleted,     // 远程删除，本地修改
}

#[derive(Debug, Clone, Serialize)]
pub enum EntityType {
    Todo,
    Plan,
    Target,
    Milestone,
    Circulation,
}

#[derive(Debug, Clone, Serialize)]
pub struct Conflict {
    pub id: String,
    pub entity_type: EntityType,
    pub entity_id: String,
    pub conflict_type: ConflictType,
    pub local_version: serde_json::Value,
    pub remote_version: serde_json::Value,
    pub local_modified_at: String,
    pub remote_modified_at: String,
}

#[derive(Debug, Clone, Serialize)]
pub enum ResolutionStrategy {
    KeepLocal,
    KeepRemote,
    Merge,
}

#[derive(Debug, Clone, Deserialize)]
pub struct Resolution {
    pub conflict_id: String,
    pub strategy: ResolutionStrategy,
    pub merged_data: Option<serde_json::Value>,
}
```

#### API 命令

```rust
// src-tauri/src/commands/sync/conflicts.rs

#[tauri::command]
pub async fn get_conflicts() -> Result<Vec<Conflict>, String> {
    // 从数据库或内存获取冲突列表
    let conflicts = detect_conflicts().await?;
    Ok(conflicts)
}

#[tauri::command]
pub async fn resolve_conflict(resolution: Resolution) -> Result<(), String> {
    // 应用解决方案
    apply_resolution(&resolution).await?;
    Ok(())
}

#[tauri::command]
pub async fn resolve_conflicts_batch(resolutions: Vec<Resolution>) -> Result<usize, String> {
    let mut count = 0;
    for resolution in resolutions {
        match apply_resolution(&resolution).await {
            Ok(_) => count += 1,
            Err(e) => log::warn!("Failed to resolve conflict: {}", e),
        }
    }
    Ok(count)
}

#[tauri::command]
pub async fn resolve_all_with_strategy(strategy: ResolutionStrategy) -> Result<usize, String> {
    let conflicts = get_conflicts().await?;
    let resolutions: Vec<Resolution> = conflicts
        .iter()
        .map(|c| Resolution {
            conflict_id: c.id.clone(),
            strategy: strategy.clone(),
            merged_data: None,
        })
        .collect();

    resolve_conflicts_batch(resolutions).await
}
```

### 4.2 组件设计

#### ConflictList 组件

```typescript
// src/components/features/ConflictList.tsx

interface ConflictListProps {
  onResolved: () => void;
}

export function ConflictList({ onResolved }: ConflictListProps) {
  const { data: conflicts, isLoading } = useQuery({
    queryKey: ['conflicts'],
    queryFn: getConflicts,
  });

  const resolveMutation = useMutation({
    mutationFn: resolveConflict,
    onSuccess: () => {
      toast.success('冲突已解决');
      onResolved();
    },
  });

  if (isLoading) return <Skeleton />;
  if (!conflicts?.length) return <EmptyState message="没有冲突" />;

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h3>同步冲突 ({conflicts.length})</h3>
        <div className="flex gap-2">
          <Button variant="outline" onClick={() => resolveAll('keep_local')}>
            全部保留本地
          </Button>
          <Button variant="outline" onClick={() => resolveAll('keep_remote')}>
            全部保留远程
          </Button>
        </div>
      </div>

      {conflicts.map((conflict) => (
        <ConflictCard
          key={conflict.id}
          conflict={conflict}
          onResolve={(strategy) =>
            resolveMutation.mutate({ conflict_id: conflict.id, strategy })
          }
        />
      ))}
    </div>
  );
}
```

```typescript
interface ConflictCardProps {
  conflict: Conflict;
  onResolve: (strategy: ResolutionStrategy) => void;
}

export function ConflictCard({ conflict, onResolve }: ConflictCardProps) {
  const [showDiff, setShowDiff] = useState(false);

  return (
    <div className="p-4 border rounded-lg">
      <div className="flex justify-between items-start mb-2">
        <div>
          <Badge>{conflict.entity_type}</Badge>
          <span className="ml-2 font-medium">

          </span>
        </div>
        <Badge variant="destructive">
          {conflict.conflict_type === 'both_modified' && '两端修改'}
          {conflict.conflict_type === 'local_deleted' && '本地已删除'}
          {conflict.conflict_type === 'remote_deleted' && '远程已删除'}
        </Badge>
      </div>

      <div className="text-sm text-muted-foreground mb-3">
        本地修改: {formatDateTime(conflict.local_modified_at)} |
        远程修改: {formatDateTime(conflict.remote_modified_at)}
      </div>

      {showDiff && (
        <div className="grid grid-cols-2 gap-4 mb-3 p-3 bg-muted rounded">
          <div>
            <h4 className="font-medium mb-2">本地版本</h4>
            <pre className="text-xs overflow-auto">
              {JSON.stringify(conflict.local_version, null, 2)}
            </pre>
          </div>
          <div>
            <h4 className="font-medium mb-2">远程版本</h4>
            <pre className="text-xs overflow-auto">
              {JSON.stringify(conflict.remote_version, null, 2)}
            </pre>
          </div>
        </div>
      )}

      <div className="flex gap-2">
        <Button size="sm" onClick={() => onResolve('keep_local')}>
          保留本地
        </Button>
        <Button size="sm" onClick={() => onResolve('keep_remote')}>
          保留远程
        </Button>
        {conflict.conflict_type === 'both_modified' && (
          <Button size="sm" variant="outline" onClick={() => onResolve('merge')}>
            合并
          </Button>
        )}
        <Button
          size="sm"
          variant="ghost"
          onClick={() => setShowDiff(!showDiff)}
        >
          {showDiff ? '隐藏详情' : '查看详情'}
        </Button>
      </div>
    </div>
  );
}
```

## 风险与缓解

| 风险           | 缓解措施                     |
| -------------- | ---------------------------- |
| 冲突检测误判   | 使用精确的时间戳和版本号对比 |
| 同步状态不一致 | 定时轮询状态，设置超时重置   |
| 并发同步冲突   | 使用锁机制防止并发同步       |
