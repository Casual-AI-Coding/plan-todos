## 功能 1: 高级搜索

### 1.1 需求描述

- 支持多条件组合搜索
- 条件包括：状态、优先级、标签、日期范围、关键词
- 支持 AND/OR 逻辑
- 实时预览搜索结果

### 1.2 数据结构

```typescript
// src/lib/types/search.ts

export type SearchOperator = "and" | "or";

export type SearchConditionType =
  | "status"
  | "priority"
  | "tag"
  | "due_date"
  | "created_at"
  | "keyword";

export interface SearchCondition {
  id: string;
  type: SearchConditionType;
  operator: SearchOperator;
  value: string | string[] | { from?: string; to?: string };
}

export interface AdvancedSearchQuery {
  conditions: SearchCondition[];
  globalOperator: "and" | "or";
}

export interface SearchResult {
  type: "todo" | "plan" | "target" | "milestone";
  id: string;
  title: string;
  snippet: string;
  highlights: { field: string; text: string }[];
  score: number;
}
```

```rust
#[derive(Debug, Deserialize)]
pub struct SearchCondition {
    pub field: String,
    pub operator: String,
    pub value: serde_json::Value,
}

#[derive(Debug, Deserialize)]
pub struct AdvancedSearchParams {
    pub conditions: Vec<SearchCondition>,
    pub global_operator: String,
    pub entity_types: Vec<String>,
    pub limit: Option<usize>,
}

#[tauri::command]
pub fn advanced_search(
    db: State<'_, Database>,
    params: AdvancedSearchParams,
) -> Result<Vec<SearchResult>, String> {
    let conn = db.lock().map_err(|e| e.to_string())?;

    let mut queries: Vec<String> = Vec::new();
    let mut all_params: Vec<Box<dyn rusqlite::ToSql>> = Vec::new();

    for condition in &params.conditions {
        let (sql, params) = build_condition_sql(condition)?;
        queries.push(sql);
        all_params.extend(params);
    }

    let joiner = if params.global_operator == "and" { " AND " } else { " OR " };
    let where_clause = queries.join(joiner);

    let sql = format!(
        "SELECT 'todo' as type, id, title, content, 1 as score FROM todos WHERE {}",
        where_clause
    );

    // Execute and return results
}

fn build_condition_sql(condition: &SearchCondition) -> Result<(String, Vec<Box<dyn rusqlite::ToSql>>), String> {
    match condition.field.as_str() {
        "status" => {
            let statuses: Vec<String> = serde_json::from_value(condition.value.clone())
                .map_err(|e| e.to_string())?;
            let placeholders = statuses.iter().map(|_| "?").collect::<Vec<_>>().join(",");
            let sql = format!("status IN ({})", placeholders);
            let params: Vec<Box<dyn rusqlite::ToSql>> = statuses.into_iter()
                .map(|s| Box::new(s) as Box<dyn rusqlite::ToSql>)
                .collect();
            Ok((sql, params))
        }
        "priority" => {
            // Similar logic
        }
        "tag" => {
            // Join with entity_tags
        }
        "due_date" => {
            // Date range query
        }
        "keyword" => {
            let keyword: String = serde_json::from_value(condition.value.clone())
                .map_err(|e| e.to_string())?;
            let sql = "(title LIKE ? OR content LIKE ?)";
            let pattern = format!("%{}%", keyword);
            let params: Vec<Box<dyn rusqlite::ToSql>> = vec![
                Box::new(pattern.clone()),
                Box::new(pattern),
            ];
            Ok((sql, params))
        }
        _ => Err(format!("Unknown field: {}", condition.field)),
    }
}
```

```typescript
// src/components/features/AdvancedSearch.tsx

interface AdvancedSearchProps {
  onSearch: (query: AdvancedSearchQuery) => void;
  onSave?: (query: AdvancedSearchQuery, name: string) => void;
}

function AdvancedSearch({ onSearch, onSave }: AdvancedSearchProps) {
  const [conditions, setConditions] = useState<SearchCondition[]>([]);
  const [globalOperator, setGlobalOperator] = useState<'and' | 'or'>('and');

  const addCondition = (type: SearchConditionType) => {
    setConditions([
      ...conditions,
      { id: generateId(), type, operator: 'and', value: '' }
    ]);
  };

  const removeCondition = (id: string) => {
    setConditions(conditions.filter((c) => c.id !== id));
  };

  const updateCondition = (id: string, updates: Partial<SearchCondition>) => {
    setConditions(
      conditions.map((c) => (c.id === id ? { ...c, ...updates } : c))
    );
  };

  return (
    <div className="space-y-4">
      {/* 全局操作符 */}
      <div className="flex gap-2">
        <Button
          variant={globalOperator === 'and' ? 'default' : 'outline'}
          onClick={() => setGlobalOperator('and')}
        >
          满足全部 (AND)
        </Button>
        <Button
          variant={globalOperator === 'or' ? 'default' : 'outline'}
          onClick={() => setGlobalOperator('or')}
        >
          满足任一 (OR)
        </Button>
      </div>

      {/* 条件列表 */}
      {conditions.map((condition, index) => (
        <SearchConditionRow
          key={condition.id}
          condition={condition}
          index={index}
          onChange={(updates) => updateCondition(condition.id, updates)}
          onRemove={() => removeCondition(condition.id)}
```

```rust
// src-tauri/src/commands/search.rs

#[tauri::command]
pub fn save_search(
    db: State<'_, Database>,
    name: String,
    query: String,
) -> Result<SavedSearch, String> {
    let conn = db.lock().map_err(|e| e.to_string())?;

    let id = uuid::Uuid::new_v4().to_string();
    let now = chrono::Utc::now().to_rfc3339();

    conn.execute(
        "INSERT INTO saved_searches (id, name, query, created_at) VALUES (?, ?, ?, ?)",
        rusqlite::params![id, name, query, now],
    ).map_err(|e| e.to_string())?;

    Ok(SavedSearch { id, name, query, created_at: now, last_used_at: None })
}

#[tauri::command]
pub fn get_saved_searches(db: State<'_, Database>) -> Result<Vec<SavedSearch>, String> {
    let conn = db.lock().map_err(|e| e.to_string())?;

    let mut stmt = conn.prepare(
        "SELECT id, name, query, created_at, last_used_at FROM saved_searches ORDER BY last_used_at DESC"
    ).map_err(|e| e.to_string())?;

    // Execute and return
}

#[tauri::command]
pub fn delete_saved_search(db: State<'_, Database>, id: String) -> Result<(), String> {
    let conn = db.lock().map_err(|e| e.to_string())?;
    conn.execute("DELETE FROM saved_searches WHERE id = ?", [id])
        .map_err(|e| e.to_string())?;
    Ok(())
}
```

### 3.2 技术设计

```typescript
// src/lib/stores/searchHistory.ts

interface SearchHistoryItem {
  query: string;
  timestamp: number;
}

const MAX_HISTORY = 20;

export const useSearchHistory = create<{ history: SearchHistoryItem[] }>()(
  persist(
    (set, get) => ({
      history: [],
      add: (query: string) => {
        const history = get().history.filter((h) => h.query !== query);
        set({
          history: [{ query, timestamp: Date.now() }, ...history].slice(
            0,
            MAX_HISTORY,
          ),
        });
      },
      remove: (query: string) => {
        set({ history: get().history.filter((h) => h.query !== query) });
      },
      clear: () => set({ history: [] }),
    }),
    { name: "plan-todos-search-history" },
  ),
);
```

```typescript
// src/components/features/GlobalSearch.tsx

function GlobalSearch() {
  const [query, setQuery] = useState('');
  const [isOpen, setIsOpen] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  // 快捷键 Ctrl+K 打开
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
        e.preventDefault();
        setIsOpen(true);
        inputRef.current?.focus();
      }
      if (e.key === 'Escape') {
        setIsOpen(false);
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  const { data: results, isLoading } = useQuery({
    queryKey: ['global-search', query],
    queryFn: () => globalSearch(query),
    enabled: query.length >= 2,
  });

  return (
    <Dialog open={isOpen} onOpenChange={setIsOpen}>
      <DialogContent className="max-w-2xl">
        <div className="relative">
          <Search className="absolute left-3 top-3 w-4 h-4 text-muted-foreground" />
          <Input
            ref={inputRef}
            placeholder="搜索 Todo、Plan、Target..."
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            className="pl-9"
          />
        </div>

        {isLoading && <Skeleton className="h-32" />}

        {results && (
          <div className="max-h-96 overflow-auto">
            {['todo', 'plan', 'target'].map((type) => {
              const items = results.filter((r) => r.type === type);
              if (!items.length) return null;

              return (
                <div key={type} className="mb-4">
                  <h4 className="text-sm font-medium text-muted-foreground mb-2">
                    {type === 'todo' ? '待办' : type === 'plan' ? '计划' : '目标'} ({items.length})
                  </h4>
                  {items.map((item) => (
                    <SearchResultItem key={item.id} item={item} highlight={query} />
                  ))}
                </div>
              );
            })}
          </div>
        )}
      </DialogContent>
    </Dialog>
  );
}

function SearchResultItem({ item, highlight }: { item: SearchResult; highlight: string }) {
  // 高亮关键词
  const highlightText = (text: string) => {
    if (!highlight) return text;
    const parts = text.split(new RegExp(`(${highlight})`, 'gi'));
    return parts.map((part, i) =>
      part.toLowerCase() === highlight.toLowerCase() ? (
        <mark key={i} className="bg-yellow-200 dark:bg-yellow-800">
          {part}
        </mark>
      ) : (
        part
      )
    );
  };

  return (
    <div
      className="p-3 hover:bg-muted rounded cursor-pointer"
      onClick={() => navigateToItem(item)}
    >
      <div className="font-medium">{highlightText(item.title)}</div>
      {item.snippet && (
        <div className="text-sm text-muted-foreground line-clamp-2">
          {highlightText(item.snippet)}
        </div>
      )}
    </div>
  );
}
```

## 风险与缓解

| 风险             | 缓解措施                       |
| ---------------- | ------------------------------ |
| 大数据量搜索慢   | 添加数据库索引，限制返回数量   |
| 复杂条件查询出错 | 分步构建 SQL，添加参数验证     |
| 搜索历史过多     | 限制最大 20 条，自动清理旧记录 |
