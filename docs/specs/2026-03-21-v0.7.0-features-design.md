### 1.1 Feature Description

- Check for new versions on app startup (max once per 24 hours)

- Support "Skip this version" option

```rust
use serde::{Deserialize, Serialize};

#[derive(Serialize, Deserialize)]
pub struct UpdateInfo {
    pub has_update: bool,
    pub current_version: String,
    pub latest_version: String,
    pub release_url: String,
    pub release_notes: String,
}

#[tauri::command]
pub async fn check_for_updates(app: tauri::AppHandle) -> Result<Option<UpdateInfo>, String> {
    // 1. Get current version from tauri.conf.json
    let current_version = app.package_info().version.to_string();

    // 2. Call GitHub API
    let client = reqwest::Client::new();
    let response = client
        .get("https://api.github.com/repos/oGsLP/plan-todos/releases/latest")
        .header("User-Agent", "plan-todos")
        .send()
        .await
        .map_err(|e| e.to_string())?;

    let release: GitHubRelease = response.json().await.map_err(|e| e.to_string())?;

    // 3. Compare versions using semver
    let latest_version = release.tag_name.trim_start_matches('v').to_string();
    let has_update = semver::Version::parse(&latest_version)
        .ok()
        .zip(semver::Version::parse(&current_version).ok())
        .map(|(latest, current)| latest > current)
        .unwrap_or(false);

    // 4. Check if skipped
    let skipped = get_skipped_version(&app)?;
    if skipped.as_deref() == Some(&latest_version) {
        return Ok(None);
    }

    Ok(Some(UpdateInfo {
        has_update,
        current_version,
        latest_version,
        release_url: release.html_url,
        release_notes: release.body.unwrap_or_default(),
    }))
}

#[tauri::command]
pub fn skip_version(app: tauri::AppHandle, version: String) -> Result<(), String> {
    // Store skipped version in app data directory
    let app_data_dir = app.path().app_data_dir().map_err(|e| e.to_string())?;
    let skip_file = app_data_dir.join(".skip_version");
    std::fs::write(&skip_file, &version).map_err(|e| e.to_string())
}
```

## Feature 2: Batch Edit

### 2.1 Feature Description

- Toggle multi-select mode in list views
- Select multiple items with checkboxes
- Batch operations toolbar at top when items selected
- Support: status, priority, tags, due_date modification
- Support: bulk delete, archive

### 2.2 Technical Design

#### State Management Hook (`src/hooks/useBatchSelect.ts`)

```typescript
import { create } from "zustand";

interface BatchSelectState {
  mode: boolean;
  selectedIds: Set<string>;
  toggleMode: () => void;
  toggle: (id: string) => void;
  selectAll: (ids: string[]) => void;
  clear: () => void;
}

export const useBatchSelect = create<BatchSelectState>((set, get) => ({
  mode: false,
  selectedIds: new Set(),
  toggleMode: () => set((s) => ({ mode: !s.mode, selectedIds: new Set() })),
  toggle: (id) =>
    set((s) => {
      const newIds = new Set(s.selectedIds);
      newIds.has(id) ? newIds.delete(id) : newIds.add(id);
      return { selectedIds: newIds };
    }),
  selectAll: (ids) => set({ selectedIds: new Set(ids) }),
  clear: () => set({ selectedIds: new Set() }),
}));
```

## Feature 3: Hotkey System

### 3.1 Feature Description

- Application-wide keyboard shortcuts (in-app only, not system-wide)
- Customizable key mappings
- Conflict detection
- Default shortcuts for common actions

### 3.2 Technical Design

#### Hotkey Provider (`src/lib/HotkeyProvider.tsx`)

```typescript
import { useEffect, useCallback } from 'react';
import { useHotkeyStore } from './useHotkeyStore';

export function HotkeyProvider({ children }: { children: React.ReactNode }) {
  const hotkeys = useHotkeyStore((s) => s.hotkeys);

  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    // Ignore if typing in input
    if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) {
      return;
    }

    const key = e.key.toLowerCase();
    const ctrl = e.ctrlKey || e.metaKey;
    const shift = e.shiftKey;
    const alt = e.altKey;

    for (const [action, binding] of Object.entries(hotkeys)) {
      if (
        binding.key.toLowerCase() === key &&
        !!binding.ctrl === ctrl &&
        !!binding.shift === shift &&
        !!binding.alt === alt
      ) {
        e.preventDefault();
        binding.callback?.();
        return;
      }
    }
  }, [hotkeys]);

  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown]);

  return <>{children}</>;
}
```

#### Hotkey Hook (`src/hooks/useHotkey.ts`)

```typescript
export function useHotkey(
  action: string,
  callback: () => void,
  deps: any[] = [],
) {
  const register = useHotkeyStore((s) => s.register);
  const unregister = useHotkeyStore((s) => s.unregister);

  useEffect(() => {
    register(action, callback);
    return () => unregister(action);
  }, [action, ...deps]);
}
```

## Feature 4: Recurring Todos

### 4.1 Feature Description

- Support advanced recurrence patterns
- Auto-create next occurrence on completion
- Support end conditions (date, count)
- Option to "detach" from recurrence chain

### 4.2 Technical Design

#### Data Model Extension

```typescript
// src/lib/types.ts

interface Recurrence {
  type: "daily" | "weekly" | "monthly" | "yearly" | "custom";
  interval: number; // Every N days/weeks/months
  daysOfWeek?: number[]; // [0-6] for weekly
  dayOfMonth?: number; // 1-31 for monthly
  weekOfMonth?: number; // 1-5 for "nth week of month"
  dayOfWeekInMonth?: number; // 0-6 for "nth weekday of month"
  endDate?: string; // ISO date string
  maxOccurrences?: number; // Max repetitions
}

interface Todo {
  // ... existing fields
  recurrence?: Recurrence;
  recurrenceFrom?: string; // Original todo ID
  recurrenceIndex?: number; // Occurrence number
}
```

#### Recurrence Service (`src/lib/services/recurrenceService.ts`)

```typescript
export function calculateNextDueDate(
  recurrence: Recurrence,
  currentDate: Date,
): Date | null {
  // Check end conditions
  if (recurrence.endDate) {
    const endDate = new Date(recurrence.endDate);
    if (currentDate >= endDate) return null;
  }

  switch (recurrence.type) {
    case "daily":
      return addDays(currentDate, recurrence.interval);

    case "weekly":
      if (recurrence.daysOfWeek) {
        return nextDayOfWeek(
          currentDate,
          recurrence.daysOfWeek,
          recurrence.interval,
        );
      }
      return addWeeks(currentDate, recurrence.interval);

    case "monthly":
      if (recurrence.weekOfMonth && recurrence.dayOfWeekInMonth !== undefined) {
        // "Nth weekday of month" pattern
        return nthWeekdayOfMonth(
          addMonths(currentDate, recurrence.interval),
          recurrence.weekOfMonth,
          recurrence.dayOfWeekInMonth,
        );
      }
      if (recurrence.dayOfMonth) {
        return setDate(
          addMonths(currentDate, recurrence.interval),
          recurrence.dayOfMonth,
        );
      }
      return addMonths(currentDate, recurrence.interval);

    case "yearly":
      return addYears(currentDate, recurrence.interval);

    default:
      return null;
  }
}

// Examples:
// Daily: { type: 'daily', interval: 1 }
// Weekdays: { type: 'weekly', daysOfWeek: [1,2,3,4,5] }
// Every 2 weeks: { type: 'weekly', interval: 2 }
// Monthly on 15th: { type: 'monthly', dayOfMonth: 15 }
// 2nd Tuesday of month: { type: 'monthly', weekOfMonth: 2, dayOfWeekInMonth: 2 }
```

#### Completion Handler

```typescript
// When marking a recurring todo as done
async function handleRecurringTodoComplete(todo: Todo) {
  if (!todo.recurrence) return;

  const nextDueDate = calculateNextDueDate(
    todo.recurrence,
    new Date(todo.due_date || Date.now()),
  );

  if (!nextDueDate) return; // Recurrence ended

  // Check occurrence count
  const nextIndex = (todo.recurrenceIndex || 0) + 1;
  if (
    todo.recurrence.maxOccurrences &&
    nextIndex > todo.recurrence.maxOccurrences
  ) {
    return; // Max occurrences reached
  }

  // Create next occurrence
  const nextTodo = {
    ...todo,
    id: generateId(),
    status: "pending",
    due_date: nextDueDate.toISOString(),
    recurrence_from: todo.recurrence_from || todo.id,
    recurrence_index: nextIndex,
    created_at: new Date().toISOString(),
    updated_at: new Date().toISOString(),
  };

  await createTodo(nextTodo);
}
```

## Feature 5: Google Drive Sync

### 5.1 Feature Description

- OAuth 2.0 authentication
- Sync data to Google Drive
- Reuse existing conflict resolution from WebDAV
- Multi-device support

### 5.2 Technical Design

#### OAuth 2.0 Flow

```
1. User clicks "Connect Google Drive"
2. Open browser with OAuth URL:
   https://accounts.google.com/o/oauth2/v2/auth?
     client_id={CLIENT_ID}&
     redirect_uri={REDIRECT_URI}&
     response_type=code&
     scope=https://www.googleapis.com/auth/drive.file&
     access_type=offline&
     prompt=consent
3. User authorizes in browser
4. Redirect to app with authorization_code
5. Exchange code for tokens:
   POST https://oauth2.googleapis.com/token
   - code, client_id, client_secret, redirect_uri, grant_type
6. Store access_token + refresh_token securely
7. Use refresh_token to get new access_token when expired
```

#### Tauri Deep Link Handler

```rust
// src-tauri/src/lib.rs
```

#### Google Drive Client (`src-tauri/src/sync/providers/google_drive.rs`)

```rust
use reqwest::Client;
use serde::{Deserialize, Serialize};

pub struct GoogleDriveClient {
    client: Client,
    access_token: String,
    refresh_token: String,
}

#[derive(Serialize, Deserialize)]
pub struct DriveFile {
    pub id: String,
    pub name: String,
    pub modified_time: String,
    pub size: u64,
}

impl GoogleDriveClient {
    pub fn new(access_token: String, refresh_token: String) -> Self {
        Self {
            client: Client::new(),
            access_token,
            refresh_token,
        }
    }

    /// Upload file to Google Drive
    pub async fn upload(&self, filename: &str, content: &[u8]) -> Result<String, Error> {
        let url = "https://www.googleapis.com/upload/drive/v3/files?uploadType=media";
        let response = self.client
            .post(url)
            .header("Authorization", format!("Bearer {}", self.access_token))
            .header("Content-Type", "application/json")
            .query(&[("name", filename)])
            .body(content.to_vec())
            .send()
            .await?;

        let file: DriveFile = response.json().await?;
        Ok(file.id)
    }

    /// Download file from Google Drive
    pub async fn download(&self, file_id: &str) -> Result<Vec<u8>, Error> {
        let url = format!("https://www.googleapis.com/drive/v3/files/{}?alt=media", file_id);
        let response = self.client
            .get(&url)
            .header("Authorization", format!("Bearer {}", self.access_token))
            .send()
            .await?;

        Ok(response.bytes().await?.to_vec())
    }

    /// List files in app folder
    pub async fn list_files(&self) -> Result<Vec<DriveFile>, Error> {
        let url = "https://www.googleapis.com/drive/v3/files";
        let response = self.client
            .get(url)
            .header("Authorization", format!("Bearer {}", self.access_token))
            .query(&[("q", "name contains 'plan-todos-sync'")])
            .send()
            .await?;

        #[derive(Deserialize)]
        struct ListResponse {
            files: Vec<DriveFile>,
        }

        let data: ListResponse = response.json().await?;
        Ok(data.files)
    }

    /// Refresh access token
    pub async fn refresh_access_token(&mut self, client_id: &str, client_secret: &str) -> Result<(), Error> {
        let url = "https://oauth2.googleapis.com/token";
        let response = self.client
            .post(url)
            .form(&[
                ("refresh_token", self.refresh_token.as_str()),
                ("client_id", client_id),
                ("client_secret", client_secret),
                ("grant_type", "refresh_token"),
            ])
            .send()
            .await?;

        #[derive(Deserialize)]
        struct TokenResponse {
            access_token: String,
        }

        let token: TokenResponse = response.json().await?;
        self.access_token = token.access_token;
        Ok(())
    }
}
```

```rust
impl SyncEngine {
    pub async fn sync_with_google_drive(&mut self) -> Result<SyncResult, Error> {
        let provider = self.get_google_drive_provider()?;

        // 1. Download remote data
        let remote_data = provider.download(SYNC_FILE_NAME).await?;
        let remote_state: SyncState = serde_json::from_slice(&remote_data)?;

        // 2. Get local changes since last sync
        let local_changes = self.get_local_changes().await?;

        // 3. Detect conflicts (reuse existing conflict detection)
        let conflicts = self.detect_conflicts(&local_changes, &remote_state);

        // 4. Resolve conflicts based on strategy
        let resolved = self.resolve_conflicts(conflicts)?;

        // 5. Merge and apply changes
        let merged_state = self.merge_states(&local_changes, &remote_state, resolved)?;

        // 6. Upload merged data
        let upload_data = serde_json::to_vec(&merged_state)?;
        provider.upload(SYNC_FILE_NAME, &upload_data).await?;

        // 7. Update sync metadata
        self.update_sync_metadata(&merged_state).await?;

        Ok(SyncResult::success())
    }
```
